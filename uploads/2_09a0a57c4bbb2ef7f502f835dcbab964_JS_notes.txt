class Bank{
   constructor(bal){
    this.currentBal=bal;
   }

   deposit(amount){
        amount
        if(amount>0){
            this.currentBal=amount+this.currentBal
            console.log(`Current Balance is ${this.currentBal}`);
        }
        else{
            console.log("Invalid Amount is Entered!");
        }
   } 

   withdraw(amount){
   
        if(amount>0 && amount<=this.currentBal){
            this.currentBal=this.currentBal-amount
            console.log(`Current Balance is ${this.currentBal}`);
        }
        else{
            console.log("Insufficient Funds");
        }
   } 
}

let b1=new Bank(8000);
b1.deposit(5000);
b1.withdraw(3000);
b1.deposit(5000);

// Inheritance
class Parent{
    constructor(name){
        this.name=name;
        console.log("Constructor Child");
      }
      
    Print() {
        console.log("Hello");
    }
    
}

class Child extends Parent{
   constructor(name){

    // to create a constructor in a child call a super constructor
    super(name);
    console.log("Constructor Child");
   }
    Print(){
        console.log("Hi")
    }
}
let obj=new Child("Mehnoor");
obj.Print();
